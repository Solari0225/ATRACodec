cmake_minimum_required(VERSION 3.20)
project(ATRACodec LANGUAGES C CXX)
option(ATRACODEC_BUILD_GUI "Build the Windows companion test GUI" ON)
option(ATRACODEC_BUILD_EXAMPLE "Build the minimal public C API example" ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
include(cmake/sources.cmake)

add_library(atracodec_objects OBJECT ${SONY_ORIGINAL_CODEC_SOURCES})
target_compile_definitions(atracodec_objects PRIVATE ${ATRACODEC_CODEC_DEFINITIONS})
set_target_properties(atracodec_objects PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(atracodec_objects PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/product")
if(MSVC)
    target_compile_options(atracodec_objects PRIVATE /fp:strict)
else()
    target_compile_options(atracodec_objects PRIVATE -fno-fast-math -ffp-contract=off -fno-math-errno)
endif()
add_library(atracodec_static STATIC $<TARGET_OBJECTS:atracodec_objects>)
add_library(atracodec_shared SHARED $<TARGET_OBJECTS:atracodec_objects>)
foreach(target atracodec_static atracodec_shared)
    set_target_properties(${target} PROPERTIES LINKER_LANGUAGE CXX)
    target_include_directories(${target} PUBLIC
        "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/product>"
        "$<INSTALL_INTERFACE:include>")
endforeach()
set_target_properties(atracodec_shared PROPERTIES OUTPUT_NAME "atracodec")
if(WIN32)
    # Keep the static library and DLL import library distinct.
    set_target_properties(atracodec_static PROPERTIES OUTPUT_NAME "atracodec-static" PREFIX "" SUFFIX ".lib")
    set_target_properties(atracodec_shared PROPERTIES PREFIX "" IMPORT_PREFIX "" IMPORT_SUFFIX ".lib" ARCHIVE_OUTPUT_NAME "atracodec-import")
    target_sources(atracodec_shared PRIVATE cmake/atracodec.def)
elseif(APPLE)
    set_target_properties(atracodec_static PROPERTIES OUTPUT_NAME "atracodec")
    set_target_properties(atracodec_shared PROPERTIES MACOSX_RPATH ON INSTALL_NAME_DIR "@rpath")
else()
    set_target_properties(atracodec_static PROPERTIES OUTPUT_NAME "atracodec")
endif()

add_executable(atracodec product/main_stream.cpp)
target_link_libraries(atracodec PRIVATE atracodec_static)
if(WIN32)
    target_link_libraries(atracodec PRIVATE shell32)
endif()
if(ATRACODEC_BUILD_EXAMPLE)
    add_executable(atracodec-c-example examples/encode_pcm16.c)
    target_link_libraries(atracodec-c-example PRIVATE atracodec_shared)
    add_executable(atracodec-roundtrip-example examples/roundtrip_memory.c)
    target_link_libraries(atracodec-roundtrip-example PRIVATE atracodec_shared)
    add_executable(atracodec-stream-example examples/stream_roundtrip.c)
    target_link_libraries(atracodec-stream-example PRIVATE atracodec_shared)
endif()

include(product/gui/cmake-native.cmake)
install(TARGETS atracodec_static atracodec_shared atracodec
    RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(FILES product/atracodec.h DESTINATION include)
install(FILES product/atracodec_stream.h DESTINATION include)
